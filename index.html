How to set up your free Google Drive sync (Takes 2 minutes):

  1. Open script.google.com and click New Project.
  2. Copy & paste the Google Apps Script code below into the editor.
  3. Click Deploy → New deployment → Select Web app.
  4. Set Execute as: Me and Who has access: Anyone.
  5. Click Deploy, authorize Google permissions, and paste the generated URL above!
Google Apps Script Code 
function doPost(e) {
  var data = e.postData.contents;
  var fileName = "ARAVIND_GYM_BACKUP.json";
  var files = DriveApp.getFilesByName(fileName);
  if (files.hasNext()) {
    files.next().setContent(data);
  } else {
    DriveApp.createFile(fileName, data, MimeType.PLAIN_TEXT);
  }
  return ContentService.createTextOutput("Saved to Drive");
}

function doGet(e) {
  var files = DriveApp.getFilesByName("ARAVIND_GYM_BACKUP.json");
  if (files.hasNext()) {
    return ContentService.createTextOutput(files.next().getBlob().getDataAsString()).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput("{}").setMimeType(ContentService.MimeType.JSON);
}
